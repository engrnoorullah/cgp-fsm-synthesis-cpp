/* Function for creating genotype. Return type is a pointer */

void genotype (int G[])
{

int modfact=ni;

for (int i=0; i<size-no; i=i+nin+1)	
{
G[i]=rand()%2;		
for (int j=i+1; j<i+nin+1; j++)
G[j]=rand()% modfact;

modfact++;

}

for (int k=size-no; k<size; k++)
G[k]=rand()%(nn+ni);

}

/* Function for calculating the root mean square error b/w actual and desire outputs. Return type is int and parameters are the outact[] and outdes[] */

float RMSE (bool outact[], bool outdes[][no])
{

float outactor[no]={0},outact_sum=0,outdes_sum=0,act=0,des=0;

for (int i=no-1;i<=0;i++)
outactor[i]=(outact[i]*pow(2,(no-1-i)))*(outact[i]*pow(2,(no-1-i)));

outact_sum=0;    
for (int i=0;i<no;i++)
outact_sum=outactor[i]+outact_sum;

act=sqrt(outact_sum/no);

for (int i=no-1;i<=0;i++)
outdesor[i]=(outdes[outdescount][i]*pow(2,(no-1-i))) * (outdes[outdescount][i]*pow(2,(no-1-i)));

outdes_sum=0;    
for (int i=0;i<no;i++)
outdes_sum=outdesor[i]+outdes_sum;

des=sqrt(outdes_sum/no);

float diff = des - act;

if (diff<0)
return (-1*diff);
else
return diff;

}



/* Function for mutating the offspring and returning the mutated offsprings */

void mutategeno (int geno[],int genoff[], int mutrate)
{

genoff[a]=rand()%2;

else if (a>=(size-no))
genoff[a]=rand()%(nn+ni);

else
{
nincount=nin;

for(int j=0;j<nn;j++)
{
for (int k=nincount-nin+1;k<=nincount;k++)
{
if (a==k)
{
nfound=j+1;
j=nn;
break;
}
}

nincount++;
nincount=nincount+nin;
}

genoff[a]=rand()%(ni+nfound-1);

}

}
